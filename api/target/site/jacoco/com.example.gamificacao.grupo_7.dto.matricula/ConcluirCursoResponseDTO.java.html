/**

 * moedasRecebidas e a quantidade creditada agora (nao o saldo).

        UUID matriculaId,
        String nomeCurso,
        CursoStatus status,
        Plano plano,
        Integer moedas,
        List<MatriculaResponseDTO> cursosLiberados,
        VoucherResponseDTO voucher,
        boolean upgradePremium,
        Integer moedasRecebidas
) {
}
